def _resolve_billing_class(
    policy: Mapping[str, Any],
    model: Any,
) -> tuple[str | None, str | None]:
    normalized_model = _norm_token(model)
    model_billing = policy.get("model_billing")
    if model_billing is not None:
        if not isinstance(model_billing, Mapping):
            return None, "trusted model_billing must be a mapping"
        normalized = {
            _norm_token(name): str(value or "").strip().upper()
            for name, value in model_billing.items()
            if isinstance(name, str)
        }
        billing = normalized.get(normalized_model)
        if billing is None:
            return None, "trusted model billing class missing for requested model"
    else:
        raw = policy.get("billing_class")
        if not isinstance(raw, str) or not raw.strip():
            return None, "trusted billing_class required for external action"
        billing = raw.strip().upper()
    if billing not in BILLING_CLASSES:
        return None, "unknown trusted billing_class"
    return billing, None

### TAKY2\enforcement.py:_model_has_capability
def _model_has_capability(
    policy: Mapping[str, Any],
    model: Any,
    capability: str,
) -> bool:
    mapping = policy.get("model_capabilities")
    if not isinstance(mapping, Mapping):
        return False
    caps = mapping.get(_norm_token(model))
    if caps is None:
        # accept original-case keys while normalizing lookup
        for name, values in mapping.items():
            if _norm_token(name) == _norm_token(model):
                caps = values
                break
    if not isinstance(caps, list):
        return False
    normalized = {str(v or "").strip().upper() for v in caps if isinstance(v, str)}
    return capability.strip().upper() in normalized

def route_gate(trusted: Mapping[str, Any], attempt: Mapping[str, Any]) -> GateResult:
    ledger = trusted.get("route_ledger", {})
    budget = ledger.get("route_budget", 3)
    if not isinstance(budget, int) or not 1 <= budget <= 3:
        return GateResult(False, "BLOCKED", "trusted route budget must be 1..3")

    if not attempt.get("target") or not attempt.get("method_or_tool"):
        return GateResult(False, "BLOCKED", "target/method required")

    route_fp = route_fingerprint(attempt)
    seen_routes = set(ledger.get("seen_routes", []))
    failed_sigs = set(ledger.get("failed_signatures", []))

    approval_evidence = []
    soft_budget_exceeded = route_fp not in seen_routes and len(seen_routes) >= budget
    if soft_budget_exceeded:
        # Route budget is an optimization signal, not an authority wall.
        # A materially different route may continue; downstream COST/KEY controls
        # still govern spend, privilege, and irreversible actions.
        approval_evidence.append("route-budget-soft-exceeded:" + str(len(seen_routes)))

    failure_by_route = ledger.get("last_failure_by_route", {})
    last = failure_by_route.get(route_fp, {}) if isinstance(failure_by_route, Mapping) else {}
    failure_class = last.get("class", "INITIAL")
    sig = stable_hash({"route": route_fp, "failure_class": failure_class})

    if failure_class in TRANSIENT_FAILURES:
        tf = ledger.get("transient_failures", {})
        if not isinstance(tf, Mapping):
            return GateResult(False, "BLOCKED", "invalid transient ledger")
        count = tf.get(route_fp, 0)
        if not isinstance(count, int):
            return GateResult(False, "BLOCKED", "invalid transient counter")
        if count >= TRANSIENT_LIMIT:
            return GateResult(
                False,
                "REROUTE_REQUIRED",
                "transient retry cap reached; choose a materially different route internally",
            )
        return GateResult(
            True,
            "ROUTE_OK",
            "bounded transient retry",
            tuple([route_fp] + approval_evidence),
        )

    if sig in failed_sigs:
        return GateResult(False, "BLOCKED", "same failed route/failure-class cannot repeat")

    capability = attempt.get("needed_capability")
    if attempt.get("ask_user") and capability in set(trusted.get("capabilities", [])):
        return GateResult(False, "USER_DEBUGGER_VIOLATION", "AI has required capability itself")

    return GateResult(
        True,
        "ROUTE_SOFT_BUDGET_EXCEEDED" if soft_budget_exceeded else "ROUTE_OK",
        (
            "materially different route allowed; soft optimization budget exceeded"
            if soft_budget_exceeded
            else "route available"
        ),
        tuple([sig, route_fp] + approval_evidence),
    )

### TAKY2\enforcement.py:_trusted_economy_decision
def _trusted_economy_decision(
    trusted: Mapping[str, Any],
    action: Mapping[str, Any],
    ref: Any,
    *,
    expected_kind: str,
    requirement_class: str | None = None,
    required_capability: str | None = None,
) -> tuple[bool, str]:
    if expected_kind not in ECONOMY_DECISION_KINDS:
        return False, "unknown economy decision kind"
    if not isinstance(ref, str) or not ref.strip():
        return False, "trusted economy decision ref missing"
    registry = trusted.get("economy_evidence_registry", {})
    item = registry.get(ref) if isinstance(registry, Mapping) else None
    if not isinstance(item, Mapping):
        return False, "economy decision is not trusted"
    if item.get("decision_kind") != expected_kind:
        return False, "economy decision kind mismatch"
    if item.get("verified") is not True:
        return False, "economy decision is not verified"
    if item.get("acceptance_state") != ACCEPTED_EVIDENCE_STATE:
        return False, "economy decision is not accepted"
    if item.get("source_status") in REJECTED_EVIDENCE_STATES:
        return False, "economy decision is stale/rejected"
    if item.get("authority_digest") != trusted.get("_authority_digest"):
        return False, "economy decision authority is stale"
    if item.get("action_digest") != action_digest(action):
        return False, "economy decision action binding mismatch"
    if item.get("semantic_owner_ref") != action.get("semantic_owner_ref"):
        return False, "economy decision semantic owner mismatch"
    if item.get("semantic_owner_scope") != action.get("semantic_owner_scope"):
        return False, "economy decision semantic owner scope mismatch"
    if requirement_class == "CAPABILITY_REQUIRED":
        expected_capability = str(required_capability or "").strip().upper()
        if not expected_capability:
            return False, "capability-required economy decision lacks required capability"
        if item.get("requirement_class") != "CAPABILITY_REQUIRED":
            return False, "economy decision requirement class mismatch"
        if str(item.get("required_capability") or "").strip().upper() != expected_capability:
            return False, "economy decision required capability mismatch"
        acceptance_ref = item.get("acceptance_ref")
        if not isinstance(acceptance_ref, str) or not acceptance_ref.strip():
            return False, "capability-required economy decision acceptance_ref missing"
    return True, ref

### TAKY2\enforcement.py:dry_run_gate
def dry_run_gate(
    trusted: Mapping[str, Any],
    action: Mapping[str, Any],
    call: Mapping[str, Any],
) -> GateResult:
    """Require a bounded execution plan before any real external call."""
    policy = action.get("call_policy")
    if not isinstance(policy, Mapping):
        return GateResult(False, "BLOCKED", "trusted action call_policy missing")

    mode = policy.get("mode")
    if mode == "LOCAL_ONLY":
        return GateResult(True, "DRY_RUN_LOCAL", "local-only action needs no external dry run")
    if mode not in {"EXTERNAL_REQUIRED", "EXTERNAL_OPTIONAL"}:
        return GateResult(False, "BLOCKED", "unknown trusted call policy mode")

    raw_plan = call.get("dry_run_plan")
    if raw_plan is None:
        plan = {}
    elif isinstance(raw_plan, Mapping):
        plan = dict(raw_plan)
    else:
        return GateResult(False, "DRY_RUN_INVALID", "dry_run_plan must be an object when provided")

    expected_calls = call.get("expected_calls", 1)
    if not isinstance(expected_calls, int) or expected_calls < 1:
        return GateResult(False, "BLOCKED", "external call expected_calls must be a positive integer")

    planned_calls = plan.get("expected_calls", expected_calls)
    if not isinstance(planned_calls, int) or planned_calls < 1:
        return GateResult(False, "DRY_RUN_INVALID", "dry_run expected_calls must be a positive integer")
    if "expected_calls" in plan and planned_calls != expected_calls:
        return GateResult(False, "DRY_RUN_MISMATCH", "explicit dry_run expected_calls conflicts with call expected_calls")

    max_calls = policy.get("max_calls", 1)
    if not isinstance(max_calls, int) or max_calls < 1:
        return GateResult(False, "BLOCKED", "trusted max_calls must be >= 1")
    effective_max_calls = 1
    if planned_calls > effective_max_calls:
        return GateResult(
            False,
            "REROUTE_LOWEST_SUFFICIENT",
            "requested call count exceeds current single-invocation executor; batch workload and rerun",
        )

    derived_fields = []
    operations = plan.get("operations")
    if operations is None:
        operation_hint = call.get("operation_hint")
        if isinstance(operation_hint, str) and operation_hint.strip():
            operations = [operation_hint.strip()]
        else:
            scope = str(action.get("execution_scope") or action.get("kind") or "external-action").strip()
            operations = [f"execute trusted action within {scope}"]
        derived_fields.append("operations")
    if not isinstance(operations, list) or not operations or not all(isinstance(x, str) and x.strip() for x in operations):
        return GateResult(False, "DRY_RUN_INVALID", "dry_run operations must be a non-empty string list")

    batch_strategy = plan.get("batch_strategy")
    if batch_strategy is None:
        batch_strategy = "single-call" if planned_calls == 1 else "bounded-batch"
        derived_fields.append("batch_strategy")
    if not isinstance(batch_strategy, str) or not batch_strategy.strip():
        return GateResult(False, "DRY_RUN_INVALID", "dry_run batch_strategy must be a string")

    fallback = plan.get("fallback", "REROUTE")
    if "fallback" not in plan:
        derived_fields.append("fallback")
    if fallback not in {"HOLD", "REROUTE", "LOCAL", "STOP"}:
        return GateResult(False, "DRY_RUN_INVALID", "dry_run fallback must be HOLD/REROUTE/LOCAL/STOP")

    billing_class, billing_error = _resolve_billing_class(policy, call.get("model"))
    if billing_error:
        return GateResult(False, "BLOCKED", billing_error)
    assert billing_class is not None
    declared_paid = plan.get("paid_api", billing_class == "PAID_API")
    if "paid_api" not in plan:
        derived_fields.append("paid_api")
    if not isinstance(declared_paid, bool):
        return GateResult(False, "DRY_RUN_INVALID", "dry_run paid_api must be boolean")
    if declared_paid != (billing_class == "PAID_API"):
        return GateResult(False, "DRY_RUN_MISMATCH", "explicit dry_run paid_api conflicts with trusted billing_class")

    execution_mode = str(policy.get("execution_mode") or "BOUNDED").strip().upper()
    if execution_mode not in {"BOUNDED", "PACKET_ONLY"}:
        return GateResult(False, "BLOCKED", "unknown trusted execution_mode")
    packet_digest = None
    if execution_mode == "PACKET_ONLY":
        if str(call.get("execution_mode") or "").strip().upper() != "PACKET_ONLY":
            return GateResult(False, "DRY_RUN_MISMATCH", "packet-only action requires PACKET_ONLY call binding")
        packet_digest = call.get("packet_digest")
        if not isinstance(packet_digest, str) or not packet_digest.strip():
            return GateResult(False, "DRY_RUN_INVALID", "packet-only action requires packet_digest")

    digest = stable_hash({
        "action_digest": action_digest(action),
        "expected_calls": planned_calls,
        "operations": operations,
        "batch_strategy": batch_strategy,
        "fallback": fallback,
        "billing_class": billing_class,
        "paid_api": declared_paid,
        "execution_mode": execution_mode,
        "packet_digest": packet_digest,
    })
    evidence = ["dry-run:" + digest]
    if raw_plan is None:
        evidence.append("dry-run-source:AUTO_DERIVED")
    elif derived_fields:
        evidence.append("dry-run-source:NORMALIZED")
    if derived_fields:
        evidence.append("dry-run-derived:" + ",".join(sorted(derived_fields)))
    return GateResult(
        True,
        "DRY_RUN_OK",
        "bounded external execution plan accepted",
        tuple(evidence),
    )

### TAKY2\enforcement.py:evaluate_savey_ai_economy
def evaluate_savey_ai_economy(
    trusted: Mapping[str, Any],
    action: Mapping[str, Any],
    call: Mapping[str, Any],
    *,
    route_fp: str | None = None,
) -> GateResult:
    policy = action.get("call_policy")
    if not isinstance(policy, Mapping):
        return GateResult(False, "BLOCKED", "trusted action call_policy missing")

    mode = policy.get("mode")
    if mode == "LOCAL_ONLY":
        return GateResult(True, "SKIP_CALL", "trusted action policy is local-only")
    if mode not in {"EXTERNAL_REQUIRED", "EXTERNAL_OPTIONAL"}:
        return GateResult(False, "BLOCKED", "unknown trusted call policy mode")

    requirement_class = str(
        policy.get("requirement_class")
        or ("QUALITY_REQUIRED" if mode == "EXTERNAL_REQUIRED" else "OPTIONAL_CONVENIENCE")
    ).strip().upper()
    if requirement_class not in {
        "CAPABILITY_REQUIRED",
        "QUALITY_REQUIRED",
        "OPTIONAL_CONVENIENCE",
    }:
        return GateResult(False, "BLOCKED", "unknown trusted requirement_class")

    required_capability = None
    if requirement_class == "CAPABILITY_REQUIRED":
        required_capability = policy.get("required_capability")
        if not isinstance(required_capability, str) or not required_capability.strip():
            return GateResult(
                False,
                "REROUTE_REQUIRED",
                "capability-required action must resolve required_capability from acceptance",
            )
        required_capability = required_capability.strip().upper()

    # Driver Necessity/Re-use/Local decisions may only reduce external work
    # when backed by a current trusted economy decision bound to this action/owner.
    skip_claims = (
        ("necessity_decision_ref", "EXTERNAL_NOT_NECESSARY", "external call is unnecessary"),
        ("reuse_decision_ref", "REUSE_SUFFICIENT", "reusable evidence/context satisfies acceptance"),
        ("local_decision_ref", "LOCAL_SUFFICIENT", "local deterministic route satisfies acceptance"),
    )
    for field, decision_kind, reason in skip_claims:
        if field in call:
            ok, detail = _trusted_economy_decision(
                trusted,
                action,
                call.get(field),
                expected_kind=decision_kind,
                requirement_class=requirement_class,
                required_capability=required_capability,
            )
            if not ok:
                return GateResult(False, "BLOCKED", detail)
            skip_evidence = ["requirement-class:" + requirement_class]
            if required_capability:
                skip_evidence.append("required-capability:" + required_capability)
            skip_evidence.append("economy-decision:" + detail)
            return GateResult(
                True,
                "SKIP_CALL",
                reason,
                tuple(skip_evidence),
            )

    # Legacy/self-asserted skip booleans are an integrity violation, not evidence.
    if call.get("necessary") is False:
        return GateResult(False, "BLOCKED", "untrusted economy necessity claim")
    if call.get("reuse_available") is True or call.get("reuse_satisfies_acceptance") is True:
        return GateResult(False, "BLOCKED", "untrusted economy reuse claim")
    if call.get("local_sufficient") is True or call.get("local_meets_acceptance") is True:
        return GateResult(False, "BLOCKED", "untrusted economy local-sufficiency claim")

    if call.get("context") == "full" and trusted.get("context", {}).get("premise_loaded"):
        if not trusted.get("context", {}).get("full_reread_authorized"):
            return GateResult(False, "SKIP_FULL_REREAD", "full context reread blocked")

    allowed_models = policy.get("allowed_models")
    if not isinstance(allowed_models, list) or not allowed_models:
        return GateResult(False, "BLOCKED", "trusted allowed_models missing")
    allowed_models = {_norm_token(v) for v in allowed_models if isinstance(v, str)}

    model = _norm_token(call.get("model"))
    if not model or model not in allowed_models:
        return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "requested model is not trusted for this action")
    if required_capability and not _model_has_capability(policy, model, required_capability):
        return GateResult(
            False,
            "REROUTE_LOWEST_SUFFICIENT",
            "requested model does not provide the trusted required capability",
        )

    model_quality = policy.get("model_quality")
    minimum_quality = policy.get("minimum_quality")
    if model_quality is not None or minimum_quality is not None:
        if not isinstance(model_quality, Mapping) or not isinstance(minimum_quality, int):
            return GateResult(False, "BLOCKED", "trusted acceptance-quality policy is malformed")
        normalized_quality = {
            _norm_token(name): score
            for name, score in model_quality.items()
            if isinstance(name, str) and isinstance(score, int)
        }
        selected_quality = normalized_quality.get(model)
        if selected_quality is None:
            return GateResult(False, "BLOCKED", "trusted quality score missing for requested model")
        if selected_quality < minimum_quality:
            return GateResult(
                False,
                "REROUTE_QUALITY_FLOOR",
                "lowest-cost route is below trusted acceptance quality floor",
            )

    expected_calls = call.get("expected_calls", 1)
    if not isinstance(expected_calls, int) or expected_calls < 1:
        return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "invalid expected_calls")

    max_calls = policy.get("max_calls", 1)
    if not isinstance(max_calls, int) or max_calls < 1:
        return GateResult(False, "BLOCKED", "trusted max_calls must be >= 1")

    billing_class, billing_error = _resolve_billing_class(policy, model)
    if billing_error:
        return GateResult(False, "BLOCKED", billing_error)
    assert billing_class is not None

    call_key = "cost:" + stable_hash(dict(call))
    approval = trusted_approval(trusted, call_key)
    approval_evidence = ["requirement-class:" + requirement_class]
    if required_capability:
        approval_evidence.append("required-capability:" + required_capability)
    approval_evidence.append("billing-class:" + billing_class)

    # SAVEY optimizes route cost after acceptance quality is protected.
    # A bounded subscription-quota call may proceed without per-call USER approval.
    # Real paid API spend, or an authority policy that explicitly demands approval,
    # still requires a trusted approval bound to this exact call.
    approval_required = (
        billing_class == "PAID_API"
        or (
            mode == "EXTERNAL_OPTIONAL"
            and policy.get("require_optional_approval") is True
        )
    )
    if approval_required:
        if not approval:
            reason = (
                "paid API call needs trusted approval"
                if billing_class == "PAID_API"
                else "optional external call policy requires trusted approval"
            )
            return GateResult(False, "USER_DECISION_REQUIRED", reason)
        approval_evidence.append("approval-key:" + call_key)

    if expected_calls > 1:
        return GateResult(
            False,
            "REROUTE_LOWEST_SUFFICIENT",
            "requested call count exceeds current single-invocation executor; batch workload and rerun",
        )

    effort = _norm_token(call.get("effort"))
    max_effort = _norm_token(policy.get("max_effort", "medium"))
    effort_rank = {"low": 1, "medium": 2, "high": 3, "xhigh": 4, "max": 5}
    if effort not in effort_rank or max_effort not in effort_rank:
        return GateResult(False, "BLOCKED", "invalid trusted/requested effort")
    if effort_rank[effort] > effort_rank[max_effort]:
        if not approval:
            return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "requested effort exceeds trusted action policy")
        if "approval-key:" + call_key not in approval_evidence:
            approval_evidence.append("approval-key:" + call_key)

    batch_size = call.get("batch_size", 1)
    if not isinstance(batch_size, int) or batch_size < 1:
        return GateResult(False, "BLOCKED", "invalid batch_size")
    if batch_size > 1:
        if policy.get("allow_batch") is not True or policy.get("batch_safe") is not True:
            return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "batching is not trusted safe for this action")

    if call.get("parallel"):
        if policy.get("allow_parallel") is not True:
            key = "parallel:" + stable_hash(dict(call))
            if not trusted_approval(trusted, key):
                return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "parallel calls are not trusted for this action")
            approval_evidence.append("approval-key:" + key)
        elif policy.get("parallel_safe") is not True or policy.get("independent_items") is not True:
            return GateResult(False, "BLOCKED", "parallel execution lacks trusted safety/independence")

    if "retry_count" in call:
        return GateResult(False, "BLOCKED", "retry_count must come from trusted route ledger")
    max_retries = policy.get("max_retries", TRANSIENT_LIMIT - 1)
    if (
        not isinstance(max_retries, int)
        or max_retries < 0
        or max_retries >= TRANSIENT_LIMIT
    ):
        return GateResult(
            False,
            "BLOCKED",
            "trusted max_retries must stay below the hard transient retry limit",
        )
    if route_fp is not None:
        ledger = trusted.get("route_ledger", {})
        transient = ledger.get("transient_failures", {}) if isinstance(ledger, Mapping) else {}
        retry_count = transient.get(route_fp, 0) if isinstance(transient, Mapping) else 0
        if not isinstance(retry_count, int) or retry_count < 0:
            return GateResult(False, "BLOCKED", "trusted transient retry ledger is invalid")
        if retry_count > max_retries:
            return GateResult(False, "THROTTLE_HOLD", "retry count exceeds trusted throttle policy")

    if call.get("subagents") and policy.get("allow_subagents") is not True:
        key = "subagents:" + stable_hash(dict(call))
        if not trusted_approval(trusted, key):
            return GateResult(False, "REROUTE_LOWEST_SUFFICIENT", "subagents are not trusted for this action")
        approval_evidence.append("approval-key:" + key)

    return GateResult(
        True,
        "COST_OK",
        "trusted action call policy satisfied",
        tuple(approval_evidence),
    )


### TAKY2\orchestrator.py:_lowest_sufficient_model
def _lowest_sufficient_model(action: Mapping[str, Any]) -> str | None:
    policy = action.get("call_policy", {})
    if not isinstance(policy, Mapping):
        return None
    allowed = policy.get("allowed_models")
    quality = policy.get("model_quality")
    minimum = policy.get("minimum_quality")
    if not isinstance(allowed, list) or not allowed:
        return None
    requirement_class = str(policy.get("requirement_class") or "").strip().upper()
    required_capability = str(policy.get("required_capability") or "").strip().upper()
    model_capabilities = policy.get("model_capabilities", {})
    if requirement_class == "CAPABILITY_REQUIRED" and not required_capability:
        return None

    def capability_ok(name: str) -> bool:
        if requirement_class != "CAPABILITY_REQUIRED":
            return True
        if not isinstance(model_capabilities, Mapping):
            return False
        values = None
        for key, candidate in model_capabilities.items():
            if str(key or "").strip().lower() == name.strip().lower():
                values = candidate
                break
        if not isinstance(values, list):
            return False
        return required_capability in {
            str(v or "").strip().upper() for v in values if isinstance(v, str)
        }

    if isinstance(quality, Mapping) and isinstance(minimum, int):
        ranked = []
        for name in allowed:
            if isinstance(name, str) and isinstance(quality.get(name), int) and capability_ok(name):
                score = quality[name]
                if score >= minimum:
                    ranked.append((score, name))
        if ranked:
            ranked.sort(key=lambda x: (x[0], x[1]))
            return ranked[0][1]
    return next(
        (v for v in allowed if isinstance(v, str) and v.strip() and capability_ok(v)),
        None,
    )

### TAKY2\orchestrator.py:_safe_correction_for_failure
def _safe_correction_for_failure(
    plan: Mapping[str, Any],
    action: Mapping[str, Any],
    gate: str,
    state: str,
    reason: str,
) -> tuple[dict[str, Any] | None, dict[str, Any] | None]:
    """Correct only execution mechanics; never amend USER intent or authority."""
    gate = str(gate or "").upper()
    state = str(state or "").upper()
    reason_l = str(reason or "").lower()
    corrected = deepcopy(dict(plan))
    context = corrected.get("context")
    attempt = corrected.get("attempt")
    call = corrected.get("call")
    if not isinstance(context, dict) or not isinstance(attempt, dict) or not isinstance(call, dict):
        return None, None

    before = deepcopy({"context": context, "attempt": attempt, "call": call})
    policy = action.get("call_policy", {})
    if not isinstance(policy, Mapping):
        policy = {}

    if gate == "COST":
        if state == "SKIP_FULL_REREAD":
            call["context"] = "delta"
        elif state in {"REROUTE_LOWEST_SUFFICIENT", "REROUTE_QUALITY_FLOOR"}:
            changed = False
            if "model" in reason_l or "quality" in reason_l:
                model = _lowest_sufficient_model(action)
                if model and call.get("model") != model:
                    call["model"] = model
                    changed = True
            if "call count" in reason_l or "expected_calls" in reason_l:
                max_calls = policy.get("max_calls", 1)
                if isinstance(max_calls, int) and max_calls >= 1:
                    # Current receipt executor performs exactly one external invocation.
                    # Batch work inside that invocation instead of emitting a multi-call receipt.
                    corrected_calls = 1
                    call["expected_calls"] = corrected_calls
                    dr = call.get("dry_run_plan")
                    if isinstance(dr, dict):
                        dr["expected_calls"] = corrected_calls
                    changed = True
            if "effort" in reason_l:
                max_effort = policy.get("max_effort", "medium")
                if isinstance(max_effort, str):
                    call["effort"] = max_effort
                    changed = True
            if "batching" in reason_l:
                call["batch_size"] = 1
                changed = True
            if "parallel" in reason_l:
                call["parallel"] = False
                changed = True
            if "subagents" in reason_l:
                call["subagents"] = False
                changed = True
            # Do not force a local substitute merely because an external route is
            # constrained. Capability/quality-required work must preserve acceptance;
            # only explicit trusted sufficiency evidence may justify reuse/local skip.
            if not changed:
                return None, None
        elif state == "THROTTLE_HOLD":
            return None, {
                "status": "HOLD_FOR_REROUTE",
                "terminal": False,
                "reason": reason,
                "next_action": "choose a materially different route; do not repeat the throttled route",
            }
        else:
            return None, None
    elif gate == "CONTEXT" and "full" in reason_l:
        context["mode"] = "delta"
    else:
        return None, None

    after = {"context": context, "attempt": attempt, "call": call}
    if before == after:
        return None, None
    correction = {
        "status": "CORRECTED",
        "terminal": False,
        "gate": gate,
        "state": state,
        "reason": reason,
        "before": before,
        "after": deepcopy(after),
        "principle": "preserve USER intent; correct execution mechanics only",
    }
    return corrected, correction

def subprocess_runner(invocation: Mapping[str, Any]) -> dict[str, Any]:
    argv = invocation.get("argv")
    if (
        not isinstance(argv, list)
        or not argv
        or not isinstance(argv[0], str)
        or not argv[0]
        or not all(isinstance(x, str) for x in argv)
    ):
        raise BridgeViolation("argv must be a string list with a non-empty executable")

    stdin_text = invocation.get("stdin_text")
    if stdin_text is not None and not isinstance(stdin_text, str):
        raise BridgeViolation("stdin_text must be a string when provided")

    timeout_raw = invocation.get("timeout_sec", NONINTERACTIVE_TIMEOUT_SEC)
    if not isinstance(timeout_raw, (int, float)) or isinstance(timeout_raw, bool):
        raise BridgeViolation("timeout_sec must be numeric")
    timeout_sec = float(timeout_raw)
    if timeout_sec <= 0 or timeout_sec > NONINTERACTIVE_TIMEOUT_MAX_SEC:
        raise BridgeViolation("timeout_sec must be > 0 and <= 300")

    cwd = invocation.get("cwd")
    if cwd is not None and (not isinstance(cwd, str) or not cwd.strip()):
        raise BridgeViolation("cwd must be a non-empty string when provided")

    env_override = invocation.get("env")
    if env_override is not None:
        if not isinstance(env_override, Mapping) or not all(
            isinstance(k, str) and isinstance(v, str) for k, v in env_override.items()
        ):
            raise BridgeViolation("env must be a string mapping when provided")

    stdin_file = None
    try:
        run_kwargs = {
            "shell": False,
            "text": True,
            "capture_output": True,
            "check": False,
            "timeout": timeout_sec,
        }
        if cwd is not None:
            run_kwargs["cwd"] = cwd

        if invocation.get("minimal_env") is True:
            if env_override is not None:
                raise BridgeViolation("minimal_env execution does not allow env overrides")
            run_kwargs["env"] = _minimal_packet_env()
        elif invocation.get("scrub_api_keys") is True or env_override is not None:
            child_env = (
                _scrubbed_external_env()
                if invocation.get("scrub_api_keys") is True
                else os.environ.copy()
            )
            if env_override is not None:
                child_env.update(dict(env_override))
            run_kwargs["env"] = child_env

        if stdin_text is None:
            run_kwargs["stdin"] = subprocess.DEVNULL
        elif invocation.get("stdin_transport") == "tempfile":
            # Some Windows model CLIs do not reliably consume a Python PIPE.
            # Use a seekable temporary file as stdin without exposing packet text
            # in argv or persisting it in the target repository.
            stdin_file = tempfile.TemporaryFile(
                mode="w+t",
                encoding="utf-8",
                newline="",
            )
            stdin_file.write(stdin_text)
            stdin_file.seek(0)
            run_kwargs["stdin"] = stdin_file
        else:
            run_kwargs["input"] = stdin_text
        completed = subprocess.run(argv, **run_kwargs)
    except subprocess.TimeoutExpired as exc:
        stdout = exc.stdout or ""
        stderr = exc.stderr or ""
        if isinstance(stdout, bytes):
            stdout = stdout.decode("utf-8", errors="replace")
        if isinstance(stderr, bytes):
            stderr = stderr.decode("utf-8", errors="replace")
        timeout_note = f"TAKY_NONINTERACTIVE_TIMEOUT after {timeout_sec:g}s"
        stderr = (str(stderr).rstrip() + "\n" + timeout_note).lstrip("\n")
        return {
            "returncode": 124,
            "stdout": str(stdout),
            "stderr": stderr,
            "timed_out": True,
            "timeout_sec": timeout_sec,
        }
    finally:
        if stdin_file is not None:
            stdin_file.close()

    return {
        "returncode": completed.returncode,
        "stdout": completed.stdout,
        "stderr": completed.stderr,
    }

### TAKY2\bridges.py:_receipt_billing_class
def _receipt_billing_class(receipt: Mapping[str, Any]) -> str | None:
    economy = receipt.get("economy_evaluation")
    if not isinstance(economy, Mapping):
        return None
    evidence = economy.get("evidence")
    if not isinstance(evidence, list):
        return None
    values = {
        item.split(":", 1)[1].strip().upper()
        for item in evidence
        if isinstance(item, str) and item.startswith("billing-class:")
    }
    if len(values) != 1:
        return None
    value = next(iter(values))
    return value if value in {"UNMETERED", "SUBSCRIPTION_QUOTA", "PAID_API"} else None

### TAKY2\bridges.py:_external_dry_run_receipt_ok
def _external_dry_run_receipt_ok(receipt: Mapping[str, Any]) -> bool:
    """Defense-in-depth: execution bridge revalidates persisted DRY_RUN evidence."""
    economy = receipt.get("economy_evaluation")
    if not isinstance(economy, Mapping):
        return False
    dry = economy.get("dry_run")
    if not isinstance(dry, Mapping) or dry.get("state") != "DRY_RUN_OK":
        return False
    gates = receipt.get("gate_receipts")
    if not isinstance(gates, list):
        return False
    for item in gates:
        if not isinstance(item, Mapping) or item.get("gate") != "COST":
            continue
        evidence = item.get("evidence")
        if isinstance(evidence, list) and any(
            isinstance(x, str) and x.startswith("dry-run:") for x in evidence
        ):
            return True
    return False

def execute_with_receipt(
    runtime: Mapping[str, Any],
    receipt_id: str,
    invocation: Mapping[str, Any],
    *,
    runner: Callable[[Mapping[str, Any]], Mapping[str, Any]] = subprocess_runner,
    executor_id: str,
) -> tuple[dict[str, Any], dict[str, Any]]:
    receipt = _receipt(runtime, receipt_id)
    if not isinstance(executor_id, str) or not executor_id.strip():
        raise BridgeViolation("executor_id required")
    if not isinstance(invocation, Mapping):
        raise BridgeViolation("invocation must be an object")

    existing_calls = runtime.get("execution_calls", {})
    if isinstance(existing_calls, Mapping) and receipt_id in existing_calls:
        raise BridgeViolation("receipt already has an executor call")

    call = invocation.get("call")
    if not isinstance(call, Mapping):
        raise BridgeViolation("invocation call binding required")
    if stable_hash(dict(call)) != receipt.get("call_digest"):
        raise BridgeViolation("actual invocation call does not match receipt")

    provider = str(invocation.get("provider") or "").strip().lower()
    model = str(call.get("model") or "").strip().lower()
    if not provider or not model:
        raise BridgeViolation("actual provider/model required")

    billing_class = None
    if receipt.get("call_allowed") is not True:
        if provider != "local" or model != "local":
            raise BridgeViolation("LOCAL_ONLY receipt cannot invoke external provider")
    else:
        if provider != "local" or model != "local":
            if not _external_dry_run_receipt_ok(receipt):
                raise BridgeViolation("external execution requires persisted DRY_RUN_OK evidence")
            billing_class = _receipt_billing_class(receipt)
            if billing_class is None:
                raise BridgeViolation("external execution requires one trusted billing-class receipt")
            expected_calls = call.get("expected_calls", 1)
            if expected_calls != 1:
                raise BridgeViolation(
                    "current receipt executor supports exactly one external invocation; "
                    "batch workload inside one invocation instead of multi-call receipt"
                )

    execution_key = receipt.get("execution_key")
    active = runtime.get("active_attempts", {}).get(execution_key)
    if not isinstance(active, Mapping) or active.get("receipt_id") != receipt_id:
        raise BridgeViolation("receipt has no matching active attempt")
    if active.get("status") != "PENDING":
        raise BridgeViolation("active attempt is not pending")

    expected_tool = str(active.get("attempt", {}).get("method_or_tool") or "").strip().lower()
    actual_tool = str(invocation.get("method_or_tool") or "").strip().lower()
    if expected_tool and actual_tool != expected_tool:
        raise BridgeViolation("actual executor route does not match receipt")

    argv_tool = _argv_identity(invocation)
    aliases = {
        "python": {"python", "python3", "py"},
        "python3": {"python", "python3", "py"},
        "py": {"python", "python3", "py"},
        "powershell": {"powershell", "pwsh"},
        "pwsh": {"powershell", "pwsh"},
        "git": {"git"},
        "claude": {"claude"},
        "codex": {"codex"},
        "codex-ephemeral": {"codex"},
        "codex-app-ephemeral": {"codex"},
        "codex-app-devnull": {"codex"},
        "gemini": {"gemini"},
    }
    if actual_tool in aliases and argv_tool and argv_tool not in aliases[actual_tool]:
        raise BridgeViolation("argv executable does not match receipt-bound route")
    if receipt.get("call_allowed") is not True and argv_tool in {"claude", "codex", "openai"}:
        raise BridgeViolation("LOCAL_ONLY receipt cannot invoke external model CLI")
    if receipt.get("call_allowed") is True and model != "local" and provider == "local":
        raise BridgeViolation("external-model receipt requires non-local provider")

    execution_mode = str(call.get("execution_mode") or "").strip().upper()
    runner_invocation = dict(invocation)
    if billing_class in {"UNMETERED", "SUBSCRIPTION_QUOTA"}:
        if invocation.get("env") is not None:
            raise BridgeViolation("subscription/unmetered route does not allow env overrides")
        runner_invocation["scrub_api_keys"] = True

    argv = invocation.get("argv")
    if model != "local" and actual_tool == "claude":
        if provider not in {"anthropic", "claude"}:
            raise BridgeViolation("Claude external route requires Anthropic provider")
        if not isinstance(argv, list) or not argv:
            raise BridgeViolation("Claude external route requires argv")
        argv_model = _argv_option_value(argv, "--model", "-m")
        if str(argv_model or "").strip().lower() != model:
            raise BridgeViolation("Claude argv model must match receipt-bound model")
        expected_effort = str(call.get("effort") or "").strip().lower()
        argv_effort = _argv_option_value(argv, "--effort")
        if expected_effort and str(argv_effort or "").strip().lower() != expected_effort:
            raise BridgeViolation("Claude argv effort must match receipt-bound effort")
    elif model != "local" and actual_tool == "codex":
        raise BridgeViolation("Codex external execution requires a dedicated hardened verifier contract")

    packet_temp_dir = None
    if execution_mode == "PACKET_ONLY":
        packet = invocation.get("stdin_text")
        if not isinstance(packet, str) or not packet:
            raise BridgeViolation("PACKET_ONLY execution requires stdin_text packet")
        if stable_hash(packet) != call.get("packet_digest"):
            raise BridgeViolation("PACKET_ONLY stdin packet digest mismatch")

        argv = invocation.get("argv")
        if not isinstance(argv, list) or not argv:
            raise BridgeViolation("PACKET_ONLY execution requires argv")

        # PACKET_ONLY is fail-closed: only explicitly hardened executors may run.
        if actual_tool != "claude" or argv_tool != "claude":
            raise BridgeViolation("PACKET_ONLY executor is not hardened/allowed")

        if provider not in {"anthropic", "claude"}:
            raise BridgeViolation("PACKET_ONLY Claude route requires Anthropic provider")

        argv_model = _argv_option_value(argv, "--model", "-m")
        if str(argv_model or "").strip().lower() != model:
            raise BridgeViolation("PACKET_ONLY Claude argv model must match receipt-bound model")
        expected_effort = str(call.get("effort") or "").strip().lower()
        argv_effort = _argv_option_value(argv, "--effort")
        if expected_effort and str(argv_effort or "").strip().lower() != expected_effort:
            raise BridgeViolation("PACKET_ONLY Claude argv effort must match receipt-bound effort")

        if invocation.get("cwd") is not None:
            raise BridgeViolation("PACKET_ONLY cwd is bridge-owned and must not be supplied")
        packet_temp_dir = tempfile.mkdtemp(prefix="TAKY2_PACKET_ONLY_")
        runner_invocation["cwd"] = packet_temp_dir

        required_flags = {
            "--no-session-persistence",
            "--strict-mcp-config",
        }
        if not required_flags.issubset(set(argv)):
            raise BridgeViolation("PACKET_ONLY Claude execution missing hardened isolation flags")
        if not any(arg in {"-p", "--print"} for arg in argv):
            raise BridgeViolation("PACKET_ONLY Claude execution requires --print")

        forbidden_flags = {
            "--add-dir",
            "--mcp-config",
            "--allowedTools",
            "--allowed-tools",
            "--plugin-dir",
            "--plugin-url",
            "--agents",
        }
        for arg in argv:
            if arg in forbidden_flags or any(
                arg.startswith(flag + "=") for flag in forbidden_flags
            ):
                raise BridgeViolation("PACKET_ONLY Claude execution contains forbidden expansion flag")

        try:
            tools_index = argv.index("--tools")
            permission_index = argv.index("--permission-prompts")
            setting_sources_index = argv.index("--setting-sources")
            settings_index = argv.index("--settings")
            output_format_index = argv.index("--output-format")
        except ValueError as exc:
            raise BridgeViolation(
                "PACKET_ONLY Claude execution must disable tools/prompts and isolate settings"
            ) from exc
        if tools_index + 1 >= len(argv) or argv[tools_index + 1] != "":
            raise BridgeViolation("PACKET_ONLY Claude execution requires empty --tools")
        if permission_index + 1 >= len(argv) or argv[permission_index + 1] != "none":
            raise BridgeViolation("PACKET_ONLY Claude execution requires --permission-prompts none")
        if setting_sources_index + 1 >= len(argv) or argv[setting_sources_index + 1] != "":
            raise BridgeViolation("PACKET_ONLY Claude execution requires empty --setting-sources")
        if settings_index + 1 >= len(argv) or argv[settings_index + 1] != "{}":
            raise BridgeViolation("PACKET_ONLY Claude execution requires empty --settings object")
        if output_format_index + 1 >= len(argv) or argv[output_format_index + 1] != "json":
            raise BridgeViolation("PACKET_ONLY Claude execution requires --output-format json")

        expected_exe = (Path.home() / ".local" / "bin" / "claude.exe").resolve()
        try:
            actual_exe = Path(argv[0]).resolve()
        except OSError as exc:
            raise BridgeViolation("PACKET_ONLY Claude executable could not be resolved") from exc
        if actual_exe != expected_exe:
            raise BridgeViolation("PACKET_ONLY Claude executable path is not pinned")
        expected_tail = [
            "-p",
            "--input-format", "text",
            "--output-format", "json",
            "--model", model,
            "--effort", expected_effort,
            "--tools", "",
            "--permission-prompts", "none",
            "--no-session-persistence",
            "--strict-mcp-config",
            "--setting-sources", "",
            "--settings", "{}",
        ]
        if argv[1:] != expected_tail:
            raise BridgeViolation("PACKET_ONLY Claude argv must match the exact hardened template")

        runner_invocation["stdin_transport"] = "tempfile"
        runner_invocation["minimal_env"] = True

    updated = deepcopy(dict(runtime))
    calls = updated.setdefault("execution_calls", {})
    if receipt_id in calls:
        raise BridgeViolation("receipt already has an executor call")
    calls[receipt_id] = {
        "executor_id": executor_id.strip(),
        "provider": provider,
        "model": model,
        "actual_calls": 1,
        "method_or_tool": actual_tool,
        "result_digest": None,
        "result": None,
        "status": "CALLING",
    }

    try:
        try:
            result = dict(runner(runner_invocation))
        except Exception as exc:
            result = {
                "returncode": 127,
                "stdout": "",
                "stderr": f"TAKY_EXECUTOR_EXCEPTION:{type(exc).__name__}:{exc}",
            }
    finally:
        if packet_temp_dir is not None:
            shutil.rmtree(packet_temp_dir, ignore_errors=True)

    result_digest = stable_hash(result)
    returncode = result.get("returncode")
    status = "RETURNED" if returncode in {None, 0} else "FAILED"
    calls[receipt_id].update({
        "result_digest": result_digest,
        "result": result,
        "status": status,
    })
    # Executor is intentionally forbidden from writing SUCCESS/completion/evidence.
    return updated, result

### TAKY2\bridges.py:record_independent_review
def record_independent_review(
    runtime: Mapping[str, Any],
    receipt_id: str,
    review: Mapping[str, Any],
    *,
    reviewer_id: str,
    review_verifier: Callable[
        [Mapping[str, Any], Mapping[str, Any], Mapping[str, Any]], bool
    ],
    review_verifier_id: str,
) -> dict[str, Any]:
    receipt = _receipt(runtime, receipt_id)
    call_record = runtime.get("execution_calls", {}).get(receipt_id)
    if not isinstance(call_record, Mapping):
        raise BridgeViolation("executor result missing before independent review")
    if call_record.get("status") != "RETURNED":
        raise BridgeViolation("independent review requires a successful returned executor call")
    if not isinstance(review, Mapping):
        raise BridgeViolation("independent review must be an object")
    if not isinstance(reviewer_id, str) or not reviewer_id.strip():
        raise BridgeViolation("independent reviewer_id required")
    if not callable(review_verifier):
        raise BridgeViolation("independent review attestation verifier required")
    if not isinstance(review_verifier_id, str) or not review_verifier_id.strip():
        raise BridgeViolation("independent review verifier_id required")
    if review_verifier(review, receipt, call_record) is not True:
        raise BridgeViolation("independent review attestation verification failed")

    orchestration = receipt.get("orchestration", {})
    if not isinstance(orchestration, Mapping):
        raise BridgeViolation("receipt orchestration binding missing")
    expected_role = orchestration.get("independent_reviewer_role")
    expected_actor = orchestration.get("independent_reviewer_actor")
    if review.get("reviewer_role") != expected_role:
        raise BridgeViolation("independent review must use receipt-bound reviewer role")
    if review.get("reviewer_actor") != expected_actor:
        raise BridgeViolation("independent review must use receipt-bound reviewer actor")

    reviewer_id = reviewer_id.strip()
    review_verifier_id = review_verifier_id.strip()
    if reviewer_id == call_record.get("executor_id"):
        raise BridgeViolation("executor cannot act as independent reviewer")
    if review_verifier_id in {reviewer_id, call_record.get("executor_id")}:
        raise BridgeViolation("independent review attestation verifier must be separate")

    evidence_ref = review.get("evidence_ref")
    source_ref = review.get("source_or_artifact_ref")
    if not isinstance(evidence_ref, str) or not evidence_ref.strip():
        raise BridgeViolation("independent review evidence_ref required")
    evidence_ref = evidence_ref.strip()
    if not isinstance(source_ref, str) or not source_ref.strip():
        raise BridgeViolation("independent review source/artifact ref required")
    if review.get("receipt_id") != receipt_id:
        raise BridgeViolation("independent review is not receipt-bound")
    if review.get("target_result_digest") != call_record.get("result_digest"):
        raise BridgeViolation("independent review target result mismatch")
    if review.get("authority_digest") != receipt.get("authority_digest"):
        raise BridgeViolation("independent review authority mismatch")

    if not isinstance(critical_gaps, list):
        raise BridgeViolation("independent review critical_gaps must be a list")
    passed = review.get("passed") is True and not critical_gaps
    acceptance_state = "ACCEPTED" if passed else "REJECTED"
    review_digest = stable_hash(dict(review))

    updated = deepcopy(dict(runtime))
    review_registry = updated.setdefault("independent_review_registry", {})
    evidence_registry = updated.setdefault("evidence_registry", {})

    }
    review_registry[evidence_ref] = review_entry
    evidence_registry[evidence_ref] = {

def verify_execution(
    runtime: Mapping[str, Any],
    receipt_id: str,
    *,
    verifier: Callable[[Mapping[str, Any], Mapping[str, Any]], Mapping[str, Any]],
    verifier_id: str,
) -> dict[str, Any]:
    receipt = _receipt(runtime, receipt_id)
    call_record = runtime.get("execution_calls", {}).get(receipt_id)
    if not isinstance(call_record, Mapping):
        raise BridgeViolation("executor result missing")
    if call_record.get("status") not in {"RETURNED", "FAILED"}:
        raise BridgeViolation("verification requires a terminal executor call")
    if not callable(verifier):
        raise BridgeViolation("independent verifier required")
    if not isinstance(verifier_id, str) or not verifier_id.strip():
        raise BridgeViolation("verifier_id required")
    if verifier_id.strip() == call_record.get("executor_id"):
        raise BridgeViolation("executor cannot self-certify")

    verdict = verifier(call_record, receipt)
    if not isinstance(verdict, Mapping):
        raise BridgeViolation("verifier verdict must be an object")

    orchestration = receipt.get("orchestration", {})
    if not isinstance(orchestration, Mapping):
        raise BridgeViolation("receipt orchestration binding missing")
    expected_verifier_role = orchestration.get("verifier_role")
    expected_verifier_actor = orchestration.get("verifier_actor")
    if verdict.get("verifier_role") != expected_verifier_role:
        raise BridgeViolation("verification must use receipt-bound verifier role")
    if verdict.get("verifier_actor") != expected_verifier_actor:
        raise BridgeViolation("verification must use receipt-bound verifier actor")

    evidence_ref = verdict.get("evidence_ref")
    evidence_type = verdict.get("evidence_type")
    if not isinstance(evidence_ref, str) or not evidence_ref:
        raise BridgeViolation("verifier evidence_ref required")
    if evidence_type not in VALID_EVIDENCE_TYPES:
        raise BridgeViolation("unsupported verifier evidence type")
    if evidence_type not in EXECUTION_EVIDENCE_TYPES:
        raise BridgeViolation("execution verifier cannot launder non-execution evidence type")

    updated = deepcopy(dict(runtime))
    outcomes = updated.setdefault("execution_outcomes", {})
    active = updated.setdefault("active_attempts", {})
    execution_key = receipt.get("execution_key")

    if verdict.get("passed") is not True:
        outcomes[receipt_id] = "VALIDATION"
        active_item = active.get(execution_key)
        scope = receipt.get("execution_scope")
        if isinstance(active_item, Mapping) and isinstance(scope, str) and scope:
            ledgers = updated.setdefault("route_ledgers", {})
            ledger = ledgers.setdefault(scope, _empty_ledger())
            ledgers[scope] = record_failure(
                ledger,
                active_item.get("attempt", {}),
                "VALIDATION",
            )
        if execution_key in active:
            active.pop(execution_key, None)
        return updated

    required = verdict.get("required_verifications", ["technical", "intent"])
    passed = verdict.get("passed_verifications", [])
    gaps = verdict.get("critical_gaps", [])
    if not isinstance(required, list) or not isinstance(passed, list) or not isinstance(gaps, list):
        raise BridgeViolation("verification lists are malformed")
    minimum = required_verifications_for_tier(
        receipt.get("verification_tier", "STANDARD")
    )
    if not minimum.issubset(set(required)):
        raise BridgeViolation("verifier under-specifies required verification tier")

    independent_review_ref = None
    independent_review = None
    if "independent" in set(required):
        expected_reviewer_role = orchestration.get("independent_reviewer_role")
        expected_reviewer_actor = orchestration.get("independent_reviewer_actor")
        if verdict.get("independent_reviewer_role") != expected_reviewer_role:
            raise BridgeViolation(
                "independent verification must use receipt-bound reviewer role"
            )
        if verdict.get("independent_reviewer_actor") != expected_reviewer_actor:
            raise BridgeViolation(
                "independent verification must use receipt-bound reviewer actor"
            )
        independent_review_ref = verdict.get("independent_review_ref")
        if not isinstance(independent_review_ref, str) or not independent_review_ref.strip():
            raise BridgeViolation("STRONG verification requires independent review receipt")
        independent_review_ref = independent_review_ref.strip()
        review_registry = runtime.get("independent_review_registry", {})
        independent_review = (
            review_registry.get(independent_review_ref)
            if isinstance(review_registry, Mapping)
            else None
        )
        if not isinstance(independent_review, Mapping):
            raise BridgeViolation("independent review receipt missing")
        if independent_review.get("receipt_id") != receipt_id:
            raise BridgeViolation("independent review receipt mismatch")
        if independent_review.get("reviewer_role") != expected_reviewer_role:
            raise BridgeViolation("independent reviewer role mismatch")
        if independent_review.get("reviewer_actor") != expected_reviewer_actor:
            raise BridgeViolation("independent reviewer actor mismatch")
        if independent_review.get("acceptance_state") != "ACCEPTED":
            raise BridgeViolation("independent review is not accepted")
        if independent_review.get("target_result_digest") != call_record.get("result_digest"):
            raise BridgeViolation("independent review target result mismatch")
        if independent_review.get("authority_digest") != receipt.get("authority_digest"):
            raise BridgeViolation("independent review authority mismatch")
        if independent_review.get("execution_scope") != receipt.get("execution_scope"):
            raise BridgeViolation("independent review scope mismatch")
        independent_reviewer_id = independent_review.get("reviewer_id")
        if not isinstance(independent_reviewer_id, str) or not independent_reviewer_id.strip():
            raise BridgeViolation("independent reviewer identity missing")
        if independent_reviewer_id.strip() == call_record.get("executor_id"):
            raise BridgeViolation("executor cannot act as independent reviewer")
        if independent_reviewer_id.strip() == verifier_id.strip():
            raise BridgeViolation("primary verifier cannot also be independent reviewer")

    if verdict.get("key_match") is not True:
        raise BridgeViolation("verifier did not confirm Key match")
    if gaps:
        raise BridgeViolation("verifier reports critical gaps")
    if set(required) - set(passed):
        raise BridgeViolation("verifier has missing required checks")

    outcomes[receipt_id] = "SUCCESS"
    if execution_key in active:
        active[execution_key]["status"] = "SUCCESS"

    evidence_refs = [evidence_ref]
    verification_evidence = {name: evidence_ref for name in passed}
    if independent_review_ref is not None:
        if independent_review_ref not in evidence_refs:
            evidence_refs.append(independent_review_ref)
        verification_evidence["independent"] = independent_review_ref

    updated.setdefault("evidence_registry", {})[evidence_ref] = {
        "type": evidence_type,
        "evidence_type": evidence_type,
        "verified": True,
        "acceptance_state": "ACCEPTED",
        "claim_supported": str(verdict.get("claim_supported") or "execution_verification"),
        "scope": receipt.get("execution_scope"),
        "producer": call_record.get("executor_id"),
        "validator": verifier_id.strip(),
        "verified_by": verifier_id.strip(),
        "verifier_role": verdict.get("verifier_role"),
        "verifier_actor": verdict.get("verifier_actor"),
        "independent_reviewer_role": verdict.get("independent_reviewer_role"),
        "independent_reviewer_actor": verdict.get("independent_reviewer_actor"),
        "independent_reviewer": verdict.get("independent_reviewer_actor"),
        "independent_review_ref": independent_review_ref,
        "source_or_artifact_ref": evidence_ref,
        "revision_or_hash": call_record.get("result_digest"),
        "authority_digest_or_key_version": receipt.get("authority_digest"),
        "semantic_owner_ref": receipt.get("semantic_owner_ref"),
        "semantic_owner_scope": receipt.get("semantic_owner_scope"),
        "verification_tier": receipt.get("verification_tier", "STANDARD"),
        "source_status": "CURRENT",
        "receipt_id": receipt_id,
        "plan_digest": receipt.get("plan_digest"),
        "action_digest": receipt.get("action_digest"),
        "call_digest": receipt.get("call_digest"),
        "result_digest": call_record.get("result_digest"),
    }
    updated.setdefault("completion_registry", {})[receipt_id] = {
        "plan_digest": receipt.get("plan_digest"),
        "action_digest": receipt.get("action_digest"),
        "call_digest": receipt.get("call_digest"),
        "authority_digest": receipt.get("authority_digest"),
        "semantic_owner_ref": receipt.get("semantic_owner_ref"),
        "semantic_owner_scope": receipt.get("semantic_owner_scope"),
        "verification_tier": receipt.get("verification_tier", "STANDARD"),
        "evidence_refs": evidence_refs,
        "verification_evidence": verification_evidence,
        "key_match": True,
        "critical_gaps": [],
        "required_verifications": list(required),
        "passed_verifications": list(passed),
        "verifier_role": verdict.get("verifier_role"),
        "verifier_actor": verdict.get("verifier_actor"),
        "independent_reviewer_role": verdict.get("independent_reviewer_role"),
        "independent_reviewer_actor": verdict.get("independent_reviewer_actor"),
        "independent_reviewer": verdict.get("independent_reviewer_actor"),
        "independent_review_ref": independent_review_ref,
        "user_confirmation_ref": None,
        "verified_by": verifier_id.strip(),
    }
    return updated

def record_economy_decision(
    runtime: Mapping[str, Any],
    action: Mapping[str, Any],
    decision: Mapping[str, Any],
    *,
    verifier: Callable[
        [Mapping[str, Any], Mapping[str, Any], Mapping[str, Any]], bool
    ],
    verifier_id: str,
) -> dict[str, Any]:
    """Record a verifier-attested SAVEY_AI_ECONOMY decision into runtime state."""
    if not isinstance(action, Mapping):
        raise BridgeViolation("economy action must be an object")
    if not isinstance(decision, Mapping):
        raise BridgeViolation("economy decision must be an object")
    if not callable(verifier):
        raise BridgeViolation("economy decision verifier required")
    if not isinstance(verifier_id, str) or not verifier_id.strip():
        raise BridgeViolation("economy decision verifier_id required")

    authority_digest = runtime.get("authority_digest")
    if not isinstance(authority_digest, str) or not authority_digest.strip():
        raise BridgeViolation("economy decision requires current authority digest")

    decision_kind = decision.get("decision_kind")
    if decision_kind not in ECONOMY_BRIDGE_DECISION_KINDS:
        raise BridgeViolation("unsupported economy decision kind")

    evidence_ref = decision.get("evidence_ref")
    source_ref = decision.get("source_or_artifact_ref")
    if not isinstance(evidence_ref, str) or not evidence_ref.strip():
        raise BridgeViolation("economy decision evidence_ref required")
    if not isinstance(source_ref, str) or not source_ref.strip():
        raise BridgeViolation("economy decision source/artifact ref required")

    owner_ref = action.get("semantic_owner_ref")
    owner_scope = action.get("semantic_owner_scope")
    if not isinstance(owner_ref, str) or not owner_ref.strip():
        raise BridgeViolation("economy action semantic_owner_ref required")
    if not isinstance(owner_scope, str) or not owner_scope.strip():
        raise BridgeViolation("economy action semantic_owner_scope required")

    policy = action.get("call_policy", {})
    if not isinstance(policy, Mapping):
        raise BridgeViolation("economy action call_policy required")
    mode = policy.get("mode")
    requirement_class = str(
        policy.get("requirement_class")
        or ("QUALITY_REQUIRED" if mode == "EXTERNAL_REQUIRED" else "OPTIONAL_CONVENIENCE")
    ).strip().upper()
    required_capability = None
    acceptance_ref = decision.get("acceptance_ref")
    if requirement_class == "CAPABILITY_REQUIRED":
        required_capability = str(policy.get("required_capability") or "").strip().upper()
        if not required_capability:
            raise BridgeViolation("capability-required economy action missing required_capability")
        if not isinstance(acceptance_ref, str) or not acceptance_ref.strip():
            raise BridgeViolation("capability-required economy decision acceptance_ref required")

    if verifier(decision, action, runtime) is not True:
        raise BridgeViolation("economy decision verification failed")

    evidence_ref = evidence_ref.strip()
    updated = deepcopy(dict(runtime))
    registry = updated.setdefault("economy_evidence_registry", {})
    if evidence_ref in registry:
        raise BridgeViolation("economy decision evidence_ref already exists")

    registry[evidence_ref] = {
        "decision_kind": decision_kind,
        "verified": True,
        "acceptance_state": "ACCEPTED",
        "source_status": "CURRENT",
        "authority_digest": authority_digest.strip(),
        "action_digest": stable_hash(dict(action)),
        "semantic_owner_ref": owner_ref.strip(),
        "semantic_owner_scope": owner_scope.strip(),
        "requirement_class": requirement_class,
        "required_capability": required_capability,
        "acceptance_ref": acceptance_ref.strip() if isinstance(acceptance_ref, str) else None,
        "source_or_artifact_ref": source_ref.strip(),
        "revision_or_hash": str(
            decision.get("revision_or_hash") or stable_hash(dict(decision))
        ),
        "verified_by": verifier_id.strip(),
        "decision_digest": stable_hash(dict(decision)),
    }
    return updated
